# AI卡牌训练平台 - 技术路线

> 详细的技术架构和实现方案，基于已确定的技术选型

## 一、技术栈总览

### 1.1 前端技术栈
```
小程序端：微信原生小程序（不使用uniapp，优先最简单实现路径）
Web游戏界面：HTML5 + Canvas + JavaScript (原生)
Canvas UI：像素化风格（Pixel Art）
UI框架：微信小程序原生组件
状态管理：小程序全局状态 / 本地存储
HTTP请求：wx.request
```

### 1.2 后端技术栈
```
运行环境：Node.js (v18+)
Web框架：Express.js
数据库：明道云（结构化数据）+ Zilliz（向量数据）
文件存储：明道云（<300G） + 宝塔服务器（>300G）
```

### 1.3 AI/ML技术栈
```
大模型：OpenAI API / Anthropic API / 本地模型
向量数据库：Zilliz Cloud
文本处理：中文分词、向量化
```

### 1.4 基础设施
```
部署：宝塔面板（前端）+ 明道云（后端）
版本控制：Git + GitHub Desktop
CI/CD：暂不考虑（手动部署）
```

### 3.1 明道云集成

#### 3.1.1 API调用封装
```javascript
// services/mingdao.js
const axios = require('axios');

class MingdaoService {
  constructor(apiKey, appKey) {
    this.apiKey = apiKey;
    this.appKey = appKey;
    this.baseURL = 'https://api.mingdao.com/v2';
  }

  // 通用请求方法
  async request(method, endpoint, data = {}) {
    const response = await axios({
      method,
      url: `${this.baseURL}${endpoint}`,
      headers: {
        'apikey': this.apiKey,
        'appkey': this.appKey,
        'Content-Type': 'application/json'
      },
      data
    });
    return response.data;
  }

  // 数据表操作
  async createRecord(tableId, data) {
    return this.request('POST', `/openapi/v2/apps/${this.appKey}/tables/${tableId}/records`, data);
  }

  async getRecords(tableId, options = {}) {
    return this.request('GET', `/openapi/v2/apps/${this.appKey}/tables/${tableId}/records`, options);
  }

  async updateRecord(tableId, recordId, data) {
    return this.request('PUT', `/openapi/v2/apps/${this.appKey}/tables/${tableId}/records/${recordId}`, data);
  }

  async deleteRecord(tableId, recordId) {
    return this.request('DELETE', `/openapi/v2/apps/${this.appKey}/tables/${tableId}/records/${recordId}`);
  }
}

module.exports = MingdaoService;
```

### 3.2 Zilliz向量数据库集成

#### 3.2.1 向量化服务
```javascript
// services/zilliz.js
const { MilvusClient } = require('@zilliz/milvus2-sdk-node');
const { OpenAIEmbeddings } = require('langchain/embeddings/openai');

class ZillizService {
  constructor(endpoint, apiKey) {
    this.client = new MilvusClient({
      address: endpoint,
      token: apiKey
    });
    this.embeddings = new OpenAIEmbeddings({
      openAIApiKey: process.env.OPENAI_API_KEY
    });
  }

  // 创建集合
  async createCollection(collectionName) {
    await this.client.createCollection({
      collection_name: collectionName,
      dimension: 1536, // OpenAI embedding维度
      metric_type: 'COSINE'
    });
  }

  // 插入向量
  async insertVector(collectionName, text, metadata) {
    // 生成向量
    const embedding = await this.embeddings.embedQuery(text);
    
    // 插入Zilliz
    await this.client.insert({
      collection_name: collectionName,
      data: [{
        id: metadata.id,
        vector: embedding,
        text: text,
        ...metadata
      }]
    });
  }

  // 向量检索
  async search(collectionName, queryText, topK = 5) {
    // 生成查询向量
    const queryVector = await this.embeddings.embedQuery(queryText);
    
    // 检索
    const results = await this.client.search({
      collection_name: collectionName,
      vector: queryVector,
      limit: topK,
      output_fields: ['text', 'gan', 'zhi', 'wuxing']
    });
    
    return results.results;
  }
}

module.exports = ZillizService;
```

### 3.3 AI服务实现

#### 3.3.1 多模型支持
```javascript
// services/ai.js
const OpenAI = require('openai');
const Anthropic = require('@anthropic-ai/sdk');

class AIService {
  constructor() {
    this.openai = new OpenAI({ apiKey: process.env.OPENAI_API_KEY });
    this.anthropic = new Anthropic({ apiKey: process.env.ANTHROPIC_API_KEY });
  }

  // 统一调用接口
  async chat(messages, options = {}) {
    const { model = 'gpt-4', provider = 'openai' } = options;
    
    if (provider === 'openai') {
      return this.chatOpenAI(messages, model);
    } else if (provider === 'anthropic') {
      return this.chatAnthropic(messages, model);
    }
  }

  async chatOpenAI(messages, model) {
    const response = await this.openai.chat.completions.create({
      model,
      messages,
      temperature: 0.7
    });
    return response.choices[0].message.content;
  }

  async chatAnthropic(messages, model) {
    const response = await this.anthropic.messages.create({
      model,
      max_tokens: 1024,
      messages
    });
    return response.content[0].text;
  }
}

module.exports = AIService;
```

#### 3.3.2 RAG实现
```javascript
// services/rag.js
const AIService = require('./ai');
const ZillizService = require('./zilliz');
const GanziService = require('./ganzi');

class RAGService {
  constructor() {
    this.ai = new AIService();
    this.zilliz = new ZillizService();
    this.ganzi = new GanziService();
  }

  async generate(query, context = {}) {
    // 1. 场景识别（地支）
    const scene = await this.ganzi.identifyScene(query);
    
    // 2. 根据天干地支检索知识
    const knowledge = await this.zilliz.search(
      'knowledge_base',
      query,
      { gan: scene.gan, zhi: scene.zhi }
    );
    
    // 3. 构建Prompt
    const prompt = this.buildPrompt(query, knowledge, context);
    
    // 4. 调用AI生成
    const response = await this.ai.chat([
      { role: 'system', content: prompt.system },
      { role: 'user', content: prompt.user }
    ], context.modelOptions);
    
    return {
      response,
      knowledge,
      scene,
      thinkingChain: this.extractThinkingChain(response)
    };
  }

  buildPrompt(query, knowledge, context) {
    const { cards, personality, role } = context;
    
    return {
      system: `你是一个${role}，性格${personality}。
      
知识库：
${knowledge.map(k => `- ${k.text}`).join('\n')}

请根据以上知识回答问题，并展示你的思考过程。`,
      user: query
    };
  }
}

module.exports = RAGService;
```

### 3.4 天干地支系统实现

#### 3.4.1 场景识别
```javascript
// services/ganzi.js
class GanziService {
  // 场景识别（对话意图 -> 地支）
  async identifyScene(query) {
    // 关键词匹配
    const keywords = {
      '子': ['分析', '战略', '决策', '市场'],
      '丑': ['目标', '计划', '行动', '制定'],
      '寅': ['优化', '升级', '迭代', '核心'],
      '卯': ['想法', '思考', '创意', '验证'],
      // ... 其他地支
    };
    
    // 匹配关键词
    for (const [zhi, words] of Object.entries(keywords)) {
      if (words.some(word => query.includes(word))) {
        return this.getZhiInfo(zhi);
      }
    }
    
    // 默认场景
    return this.getZhiInfo('子');
  }

  // 获取地支信息（含藏干）
  getZhiInfo(zhi) {
    const zhiMap = {
      '子': { gan: '癸', canggan: { '癸': 1.0 } },
      '丑': { gan: '己', canggan: { '己': 0.6, '癸': 0.3, '辛': 0.1 } },
      '寅': { gan: '甲', canggan: { '甲': 0.6, '丙': 0.3, '戊': 0.1 } },
      // ... 其他地支
    };
    return zhiMap[zhi] || zhiMap['子'];
  }

  // 五行相生相克判断
  checkWuxing(wuxing1, wuxing2) {
    const sheng = { '木': '火', '火': '土', '土': '金', '金': '水', '水': '木' };
    const ke = { '木': '土', '土': '水', '水': '火', '火': '金', '金': '木' };
    
    return {
      sheng: sheng[wuxing1] === wuxing2,
      ke: ke[wuxing1] === wuxing2
    };
  }
}

module.exports = GanziService;
```

## 四、Canvas像素化UI实现

### 4.1 像素化设计原则
- **图像格式**：使用PNG格式，支持透明
- **分辨率**：低分辨率像素图（16x16, 32x32, 64x64）
- **颜色**：限制调色板，减少文件大小
- **动画**：使用精灵图（Sprite Sheet）

### 4.2 Canvas渲染优化
```javascript
// canvas.js - 像素化渲染引擎
class PixelCanvas {
  constructor(canvasId) {
    this.canvas = document.getElementById(canvasId);
    this.ctx = this.canvas.getContext('2d');
    // 禁用抗锯齿，保持像素化效果
    this.ctx.imageSmoothingEnabled = false;
  }

  // 绘制像素化精灵
  drawSprite(sprite, x, y, scale = 1) {
    this.ctx.save();
    this.ctx.imageSmoothingEnabled = false;
    this.ctx.drawImage(
      sprite,
      x, y,
      sprite.width * scale,
      sprite.height * scale
    );
    this.ctx.restore();
  }

  // 批量渲染优化
  renderBatch(sprites) {
    // 使用离屏Canvas提高性能
    const offscreen = document.createElement('canvas');
    const offCtx = offscreen.getContext('2d');
    offCtx.imageSmoothingEnabled = false;
    
    sprites.forEach(sprite => {
      offCtx.drawImage(sprite.image, sprite.x, sprite.y);
    });
    
    this.ctx.drawImage(offscreen, 0, 0);
  }
}

// pixel-art.js - 像素化资源管理
class PixelArtManager {
  constructor() {
    this.sprites = new Map();
    this.loaded = false;
  }

  // 加载像素化资源
  async loadSprites() {
    const spritePaths = [
      '/assets/sprites/card-background.png',
      '/assets/sprites/character-idle.png',
      '/assets/sprites/ui-elements.png'
    ];

    for (const path of spritePaths) {
      const img = new Image();
      img.src = path;
      await new Promise((resolve) => {
        img.onload = () => {
          this.sprites.set(path, img);
          resolve();
        };
      });
    }
    this.loaded = true;
  }

  // 获取精灵图
  getSprite(path) {
    return this.sprites.get(path);
  }
}
```

### 4.3 性能优化
- **资源预加载**：游戏开始前预加载所有像素化资源
- **对象池**：复用Canvas对象，减少内存分配
- **分层渲染**：背景层、游戏层、UI层分离
- **脏矩形**：只重绘变化区域
- **离屏Canvas**：复杂场景使用离屏Canvas预渲染

### 4.4 内存管理
- **图片压缩**：像素图本身文件小（通常<50KB）
- **及时释放**：不用的资源及时释放
- **缓存策略**：常用资源缓存，不常用及时清理
- **资源限制**：单个场景资源总量<2MB
